# План добавления Telegram MiniApp в бота записи на консультацию

## 📋 Обзор

Добавление Telegram MiniApp (WebApp) для записи на консультацию с календарём, выбором специалиста и слотов. MVP включает базовый UI с календарём, выпадающим списком специалистов и временными слотами.

### 2.2 `webapp/auth.py` — Аутентификация через Telegram WebApp
```python
import hashlib
import hmac
import time
from fastapi import HTTPException, Header
from config.settings import settings

def verify_telegram_init_data(init_data: str) -> dict:
    """Верификация Telegram WebApp init data."""
    try:
        data = parse_query_string(init_data)
        check_hash = data.get('hash', '')
        del data['hash']
        
        auth_date = int(data.get('auth_date', 0))
        if time.time() - auth_date > 86400:  # 24 часа
            raise HTTPException(status_code=401, detail="Data is expired")
        
        # Проверка hash
        secret = hmac.new(b"WebAppData", settings.bot_token.encode(), hashlib.sha256).digest()
        data_check = "&".join(f"{k}={v}" for k, v in sorted(data.items()))
        calculated_hash = hmac.new(secret, data_check.encode(), hashlib.sha256).hexdigest()
        
        if calculated_hash != check_hash:
            raise HTTPException(status_code=401, detail="Invalid hash")
        
        return data.get('user', {})
    except Exception as e:
        raise HTTPException(status_code=401, detail=str(e))
```

### 2.3 `webapp/routes/slots.py` — Получение слотов
```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from webapp.dependencies import get_db_session
from database.repositories import get_available_slots
from datetime import datetime

router = APIRouter()

@router.get("/slots")
async def get_slots(
    specialist_id: int,
    date: str,
    db: AsyncSession = Depends(get_db_session)
):
    """Получение доступных слотов для специалиста и даты."""
    try:
        target_date = datetime.strptime(date, "%Y-%m-%d")
    except ValueError:
        raise HTTPException(status_code=400, detail="Invalid date format")
    
    slots = await get_available_slots(db, specialist_id, target_date)
    return {"slots": slots}
```

### 2.4 `webapp/routes/bookings.py` — Создание бронирования
```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from webapp.dependencies import get_db_session
from webapp.schemas import BookingCreate
from services.booking_service import BookingService

router = APIRouter()

@router.post("")
async def create_booking(
    data: BookingCreate,
    user_data: dict = Depends(verify_telegram_auth),
    db: AsyncSession = Depends(get_db_session)
):
    """Создание бронирования из MiniApp."""
    result = await BookingService.create_booking(
        db=db,
        user_id=int(user_data['id']),
        telegram_username=user_data.get('username'),
        specialist_id=data.specialist_id,
        date_str=data.date,
        time_str=data.time,
        slot_key=f"{data.specialist_id}_{data.date}_{data.time}",
        name=data.name,
        phone=data.phone,
        additional_info=data.additional_info,
    )
    return {"booking": result}
```

### 3.2 `app/src/components/Calendar.tsx` — Календарь
```tsx
import { useState } from 'react'
import { useBookingStore } from '../store/bookingStore'
import api from '../api/client'

export default function Calendar() {
  const { specialist, setDate, nextStep } = useBookingStore()
  const [dates, setDates] = useState<Date[]>([])
  
  // Загрузка доступных дат
  useEffect(() => {
    loadAvailableDates()
  }, [specialist])
  
  const loadAvailableDates = async () => {
    // Запрос к API для получения доступных дат
    const response = await api.get(`/availability?specialist_id=${specialist.id}`)
    setDates(response.data.dates)
  }
  
  const handleDateSelect = (date: Date) => {
    setDate(date.toISOString().split('T')[0])
    nextStep()
  }
  
  return (
    <div className="p-4">
      <h2 className="text-lg font-semibold mb-4">Выберите дату</h2>
      <div className="grid grid-cols-7 gap-2">
        {dates.map((date) => {
          const isAvailable = date.hasSlots
          return (
            <button
              key={date.toISOString()}
              onClick={() => isAvailable && handleDateSelect(date)}
              className={`p-3 rounded-lg text-center ${
                isAvailable
                  ? 'bg-blue-500 text-white hover:bg-blue-600'
                  : 'bg-gray-300 text-gray-500 cursor-not-allowed'
              }`}
              disabled={!isAvailable}
            >
              {date.getDate()}
            </button>
          )
        })}
      </div>
    </div>
  )
}
```

### 3.3 `app/src/components/TimeSlots.tsx` — Временные слоты
```tsx
import { useEffect, useState } from 'react'
import { useBookingStore } from '../store/bookingStore'
import api from '../api/client'

export default function TimeSlots() {
  const { specialist, date, setTime, nextStep } = useBookingStore()
  const [slots, setSlots] = useState<any[]>([])
  
  useEffect(() => {
    loadSlots()
  }, [specialist, date])
  
  const loadSlots = async () => {
    const response = await api.get(`/slots?specialist_id=${specialist.id}&date=${date}`)
    setSlots(response.data.slots)
  }
  
  const handleSlotSelect = (slot: any) => {
    setTime(slot.time)
    nextStep()
  }
  
  return (
    <div className="p-4">
      <h2 className="text-lg font-semibold mb-4">Выберите время</h2>
      <div className="grid grid-cols-3 gap-2">
        {slots.map((slot) => (
          <button
            key={slot.time}
            onClick={() => handleSlotSelect(slot)}
            className={`p-3 rounded-lg text-center ${
              slot.available
                ? 'bg-blue-500 text-white hover:bg-blue-600'
                : 'bg-gray-300 text-gray-500 cursor-not-allowed'
            }`}
            disabled={!slot.available}
          >
            {slot.time}
          </button>
        ))}
      </div>
    </div>
  )
}
```

```tsx
export default function SpecialistSelect() {
```

### 3.6 `app/src/store/bookingStore.ts` — State management
```typescript
import { create } from 'zustand'

type BookingState = {
  specialist: any
  date: string
  time: string
  step: 'specialist' | 'calendar' | 'slots' | 'form' | 'confirmation'
  setSpecialist: (spec: any) => void
  setDate: (date: string) => void
  setTime: (time: string) => void
  nextStep: () => void
  reset: () => void
}

export const useBookingStore = create<BookingState>((set) => ({
  specialist: null,
  date: '',
  time: '',
  step: 'specialist',
  setSpecialist: (spec) => set({ specialist: spec }),
  setDate: (date) => set({ date }),
  setTime: (time) => set({ time }),
  nextStep: () => set((state) => ({
    step: state.step === 'specialist' ? 'calendar' :
          state.step === 'calendar' ? 'slots' :
          state.step === 'slots' ? 'form' :
          state.step === 'form' ? 'confirmation' : state.step
  })),
  reset: () => set({ specialist: null, date: '', time: '', step: 'specialist' })
}))
```

## ⚠️ Важные замечания

1. **Аутентификация:** MiniApp получает данные пользователя из Telegram WebApp init data. Необходимо верифицировать hash на бэкенде.
2. **CORS:** Настроить CORS для разрешённых origins.
3. **HTTPS:** Telegram требует HTTPS для WebApp.
4. **Slot locking:** Использовать существующий механизм `SlotLock` для защиты от двойной записи.
5. **Consent:** Сохранять согласие на ПДН, как в текущем боте.
6. **Google Sheets:** Записи из MiniApp также попадают в Google Sheets через `SheetService`.
